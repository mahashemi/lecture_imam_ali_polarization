\section{Opening}

\begin{frame}{Love and Hate Are Not Opposites --- They're One Faith}
\footnotesize
\bigquote{Is belief anything but love and hate?}{Imam al-Sadiq (a.s.), Al-K\=afi, Bk.\ 1, Ch.\ 60, Hadith 5}
\bigquote{The strongest ring of belief to hold onto\ldots\ is love for the sake of Allah, hate for the sake of Allah.}{The Prophet (s), Al-K\=afi, Bk.\ 1, Ch.\ 60, Hadith 6}
\begin{itemize}[itemsep=2pt]
  \item Faith made \emph{beloved}. Disbelief made \emph{hateful}. (Qur'an 49:7)
  \item Not prayer alone. Not fasting alone. Not hajj alone.
  \item Imam al-Baqir (a.s.): without love/hate on the basis of religion, one has, in effect, no religion.
\end{itemize}
\examplenote{The companions guess}{They suggest prayer, zakat, fasting, hajj, jihad -- all wrong; the Prophet corrects them: it is love and hate for the sake of Allah.}
\delivernote{Show of hands: who has heard the phrase ``love for the sake of Allah, hate for the sake of Allah'' before tonight?}

\end{frame}

% ---------------- Q&A ----------------
\qaframe{Opening Hadiths}{%
  \qa{According to the hadith, what did Imam al-Sadiq say belief actually consists of?}
     {He said belief is nothing \emph{but} love and hate -- then recited Qur'an 49:7, where faith is made beloved to the heart and disbelief is made hateful to it.}
  \qa{What did the Prophet identify as the strongest handhold of belief?}
     {Not prayer, fasting, or hajj alone, but loving for the sake of Allah, hating for the sake of Allah, and aligning one's loyalty and rejection accordingly.}
}
